# VJN6 plugin (VST3, AU) and standalone app (SPEC §9.2, §9.3). Included by the root CMakeLists.txt
# when VJN6_BUILD_PLUGIN is ON, after add_subdirectory(third_party/JUCE).

# The UI bundle served by the editor (§9.4: one Svelte bundle for all builds). Built by
# `npm run build` at the repository root.
set(VJN6_UI_DIST "${PROJECT_SOURCE_DIR}/ui/dist" CACHE PATH "Built UI bundle (index.html and assets) embedded into the plugin")
if(NOT EXISTS "${VJN6_UI_DIST}/index.html")
  message(WARNING "VJN6 plugin and standalone skipped: ${VJN6_UI_DIST}/index.html not found. "
                  "Build the UI first (npm ci && npm run build at the repository root), then re-run CMake.")
  return()
endif()

juce_add_plugin(VJN6
    PRODUCT_NAME "VJN6"  PLUGIN_NAME "VJN6"  VERSION 0.1.0
    COMPANY_NAME "Marco Funk"  COMPANY_COPYRIGHT "© 2026 Marco Funk"
    BUNDLE_ID com.marcofunk.vjn6
    PLUGIN_MANUFACTURER_CODE Mfnk  PLUGIN_CODE Vjn6
    DESCRIPTION "6-voice virtual-analog synthesizer"
    FORMATS Standalone VST3 AU  IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE  NEEDS_MIDI_OUTPUT TRUE  IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    VST3_CATEGORIES Instrument Synth  AU_MAIN_TYPE kAudioUnitType_MusicDevice
    NEEDS_WEBVIEW2 TRUE  NEEDS_WEB_BROWSER TRUE
    MICROPHONE_PERMISSION_ENABLED FALSE  HARDENED_RUNTIME_ENABLED TRUE
    COPY_PLUGIN_AFTER_BUILD TRUE)

# ---- embedded UI: every file of the bundle, staged as ui<N>.<ext>; VJN6UiPaths.h maps URL paths ----
file(GLOB_RECURSE vjn6_ui_files LIST_DIRECTORIES false CONFIGURE_DEPENDS "${VJN6_UI_DIST}/*")
list(SORT vjn6_ui_files)
set(vjn6_ui_staged)
set(vjn6_ui_table "")
set(vjn6_ui_index 0)
foreach(src IN LISTS vjn6_ui_files)
  get_filename_component(base "${src}" NAME)
  if(base STREQUAL ".DS_Store")
    continue()
  endif()
  file(RELATIVE_PATH rel "${VJN6_UI_DIST}" "${src}")
  get_filename_component(ext "${src}" LAST_EXT)
  set(staged "ui${vjn6_ui_index}${ext}")
  set(dst "${CMAKE_CURRENT_BINARY_DIR}/ui_staged/${staged}")
  add_custom_command(OUTPUT "${dst}"
    COMMAND ${CMAKE_COMMAND} -E copy "${src}" "${dst}"
    DEPENDS "${src}"
    VERBATIM)
  list(APPEND vjn6_ui_staged "${dst}")
  string(REPLACE "\\" "\\\\" rel_c "${rel}")
  string(REPLACE "\"" "\\\"" rel_c "${rel_c}")
  string(APPEND vjn6_ui_table "    {\"${rel_c}\", \"${staged}\"},\n")
  math(EXPR vjn6_ui_index "${vjn6_ui_index} + 1")
endforeach()
juce_add_binary_data(VJN6UiData NAMESPACE VJN6UiData HEADER_NAME VJN6UiData.h SOURCES ${vjn6_ui_staged})
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/VJN6UiPaths.h" CONTENT
"// Generated by plugin/CMakeLists.txt: URL path of each embedded UI file -> its staged file name.
#pragma once
namespace vjn6plugin {
struct UiPath { const char* url; const char* staged; };
inline constexpr UiPath kUiPaths[] = {
${vjn6_ui_table}};
}  // namespace vjn6plugin
" @ONLY)

set(VJN6_PLUGIN_SOURCES
  ${CMAKE_CURRENT_SOURCE_DIR}/source/HostParams.cpp
